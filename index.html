    const cleanUrl = (url || "").trim();

    if (!cleanUrl) {
      bgLayer.style.backgroundImage = "";
      if (bgStatusMsg) bgStatusMsg.textContent = "Đang dùng: Bảng Hồ Sơ Mặc Định";
      if (save) localStorage.removeItem("showrunner_bg_url");
      return;
    }

    bgLayer.style.backgroundImage = `url("${cleanUrl.replace(/"/g, '\\"')}")`;
    if (bgUrlInput && !cleanUrl.startsWith("data:")) bgUrlInput.value = cleanUrl;
    if (bgStatusMsg) {
      bgStatusMsg.textContent = cleanUrl.startsWith("data:")
        ? "Đã áp dụng ảnh từ máy của bạn"
        : "Đã áp dụng ảnh từ link URL";
    }
    if (save) {
      try { localStorage.setItem("showrunner_bg_url", cleanUrl); } catch (e) {}
    }
  }
